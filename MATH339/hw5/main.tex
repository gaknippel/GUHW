\section*{Problem 37}

\[
c_1\mathbf{w}_1 + c_2\mathbf{w}_2 + c_3\mathbf{w}_3 = \mathbf{0}
\]

\[
c_1(\mathbf{v}_1 + \mathbf{v}_2) + c_2(\mathbf{v}_2 - \mathbf{v}_3) + c_3(\mathbf{v}_2 + \mathbf{v}_3) = \mathbf{0}
\]

\[
c_1\,\mathbf{v}_1 + (c_1 + c_2 + c_3)\,\mathbf{v}_2 + (-c_2 + c_3)\,\mathbf{v}_3 = \mathbf{0}
\]

$S$ linearly independent $\Longrightarrow$
\[
\begin{aligned}
c_1 &= 0 \\
c_1 + c_2 + c_3 &= 0 \\
-c_2 + c_3 &= 0
\end{aligned}
\]

\[
c_3 = c_2 \;\Longrightarrow\; 2c_2 = 0 \;\Longrightarrow\; c_1 = c_2 = c_3 = 0
\]

$so  T$ is linearly independent.
